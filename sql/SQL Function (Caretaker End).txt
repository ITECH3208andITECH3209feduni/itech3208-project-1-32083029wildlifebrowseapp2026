-- For use in the Loops to define what a plant_id and quantity input go into
CREATE TYPE delivery_item_type AS (
    plant_id INT,
    quantity SMALLINT
);



-- Procedure to Insert Delivery Items into Specific Request (For Editing maybe)
CREATE OR REPLACE FUNCTION insert_delivery_items(
    in_delivery_id INT,
    in_items delivery_item_type[]
) RETURNS VOID AS $$
DECLARE
    item delivery_item_type;
BEGIN
    -- Loop over each item in the input array
    FOREACH item IN ARRAY in_items
    LOOP
        INSERT INTO DeliveryItem (delivery_ID, plant_ID, quantity)
        VALUES (in_delivery_id, item.plant_id, item.quantity);
    END LOOP;
END;
$$ LANGUAGE plpgsql;


-- Example Using Procedure DeliveryRequest additions (Edit a Request Potential?)

SELECT insert_delivery_items(
    -- Inserts into DeliveryRequest with ID 1
    1,
    ARRAY[
        -- Adds a Item of type Eculyptus and M3 of 15 
	ROW(1, 15)::delivery_item_type,
        ROW(2, 10)::delivery_item_type
    ]
);



-- Input details of Delivery with Items inside it.

CREATE OR REPLACE FUNCTION create_full_delivery(
    in_name VARCHAR,
    in_address VARCHAR,
    in_specifications VARCHAR,
    in_animal_id INT,
    in_items delivery_item_type[]
) RETURNS INTEGER
LANGUAGE plpgsql
AS $$
DECLARE
    new_delivery_id INTEGER;
    item delivery_item_type;
BEGIN
    -- Insert the Delivery Request and capture its ID
    INSERT INTO DeliveryRequest (name, address, specifications,animal_ID)
    VALUES (in_name, in_address, in_specifications,in_animal_ID)
    RETURNING delivery_ID INTO new_delivery_id;

    -- Loop over each item and insert into DeliveryItem
    FOREACH item IN ARRAY in_items
    LOOP
        INSERT INTO DeliveryItem (delivery_ID, plant_ID, quantity)
        VALUES (new_delivery_id, item.plant_id, item.quantity);
    END LOOP;

    -- Basic return for delivery_ID (For Refrence)
    RETURN new_delivery_id;
END;
$$;


-- Example Create Full DeliveryRequest + DeliveryItems 

SELECT create_full_delivery(
    'Wildlife Park',
    '22 Leaf Street, Ballarat VIC',
    'Koala daily feed',
    1,
    ARRAY[
        ROW(1, 20)::delivery_item_type,  -- Eucalyptus, 20
        ROW(2, 15)::delivery_item_type   -- Silverbeet, 15
    ]
);
