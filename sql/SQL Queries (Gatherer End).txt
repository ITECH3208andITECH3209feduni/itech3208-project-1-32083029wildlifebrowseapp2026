SELECT 
    dr.delivery_ID,
    dr.name AS requester_name,
    dr.address,
    dr.postcode,
    dr.specifications,
    GROUP_CONCAT(CONCAT(p.plant_Name, ' x', di.quantity) SEPARATOR ', ') AS items_delivered
FROM 
    DeliveryRequest dr
JOIN 
    DeliveryItem di ON dr.delivery_ID = di.delivery_ID
JOIN 
    Plant p ON di.plant_ID = p.plant_ID
GROUP BY 
    dr.delivery_ID, dr.name, dr.address, dr.specifications
ORDER BY 
    dr.delivery_ID;




-- Basic JSON output for all deliveries (Will be broken into standard JSON formatting, with al appropiate names) 


SELECT jsonb_build_object(
  'delivery_ID', dr.delivery_ID,
  'name', dr.name,
  'address', dr.address,
  'postcode' dr.postcode,
  'specifications', dr.specifications,
  'animal', jsonb_build_object(
      'animal_ID', a.animal_ID,
      'animal_Name', a.animal_Name
  ),
  'items', jsonb_agg(
    jsonb_build_object(
      'plant_ID', p.plant_ID,
      'plant_Name', p.plant_Name,
      'quantity', di.quantity
    )
  )
) AS delivery_json
FROM DeliveryRequest dr
LEFT JOIN Animal a ON dr.animal_ID = a.animal_ID
JOIN DeliveryItem di ON dr.delivery_ID = di.delivery_ID
JOIN Plant p ON di.plant_ID = p.plant_ID
GROUP BY dr.delivery_ID, a.animal_ID;
