CREATE TABLE Plant (
    plant_ID SERIAL PRIMARY KEY,
    plant_Name VARCHAR(40) NOT NULL
);
-- Table Animals
CREATE TABLE Animal (
    animal_ID SERIAL PRIMARY KEY,
    animal_Name VARCHAR(40) NOT NULL
);
-- DeliveryRequest Table
CREATE TABLE DeliveryRequest (
    delivery_ID SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    address VARCHAR(100) NOT NULL,
    postcode VARCHAR(4),
    specifications VARCHAR(200) NOT NULL,
    animal_ID INT NULL,
    FOREIGN KEY (animal_ID) REFERENCES Animal(animal_ID) ON DELETE SET NULL
);

-- DeliveryItem Table
CREATE TABLE DeliveryItem (
    item_ID SERIAL PRIMARY KEY,
    delivery_ID INT NOT NULL,
    plant_ID INT NOT NULL,
    quantity SMALLINT CHECK (quantity >= 1),
    FOREIGN KEY (delivery_ID) REFERENCES DeliveryRequest(delivery_ID) ON DELETE CASCADE,
    FOREIGN KEY (plant_ID) REFERENCES Plant(plant_ID)
);
